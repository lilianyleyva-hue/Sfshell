// plantilla: par_impar
// titulo: Programa que dice si un número es par o impar
// descripcion: Lee un número entero y dice si es par o impar (también con negativos).
// palabras: programa, par, impar, numero, entero, leer, decir, saber
// caso: {"huecos": {}, "caso": {"entrada": "4\n", "esperado": "El 4 es par.", "comparar": "contiene"}}
// caso: {"huecos": {}, "caso": {"entrada": "7\n", "esperado": "El 7 es impar.", "comparar": "contiene"}}
// caso: {"huecos": {}, "caso": {"entrada": "-3\n", "esperado": "El -3 es impar.", "comparar": "contiene"}}
// caso: {"huecos": {}, "caso": {"entrada": "0\n", "esperado": "El 0 es par.", "comparar": "contiene"}}
// caso: {"huecos": {}, "caso": {"entrada": "hola\n", "esperado": "Eso no es un número entero.", "comparar": "contiene"}}
package main

import (
	"bufio"
	"fmt"
	"os"
	"strconv"
	"strings"
)

func main() {
	lector := bufio.NewReader(os.Stdin)
	fmt.Print("Escribe un número entero: ")
	linea, _ := lector.ReadString('\n')
	n, err := strconv.Atoi(strings.TrimSpace(linea))
	if err != nil {
		fmt.Println("Eso no es un número entero.")
		return
	}
	if n%2 == 0 {
		fmt.Printf("El %d es par.\n", n)
	} else {
		fmt.Printf("El %d es impar.\n", n)
	}
}

// plantilla: mayor_de_tres
// titulo: Programa que dice el mayor de tres números
// descripcion: Pide tres números y escribe cuál es el mayor.
// palabras: programa, mayor, maximo, tres, numero, pedir, leer, comparar
// caso: {"huecos": {}, "caso": {"entrada": "3\n9\n4\n", "esperado": "El mayor es 9", "comparar": "contiene"}}
// caso: {"huecos": {}, "caso": {"entrada": "-1\n-5\n-2\n", "esperado": "El mayor es -1", "comparar": "contiene"}}
// caso: {"huecos": {}, "caso": {"entrada": "2,5\n2\n1\n", "esperado": "El mayor es 2.5", "comparar": "contiene"}}
// caso: {"huecos": {}, "caso": {"entrada": "7\n7\n7\n", "esperado": "El mayor es 7", "comparar": "contiene"}}
package main

import (
	"bufio"
	"fmt"
	"os"
	"strconv"
	"strings"
)

// leerNumero pregunta hasta que le dan un número (acepta coma o punto decimal).
func leerNumero(lector *bufio.Reader, pregunta string) float64 {
	for {
		fmt.Print(pregunta)
		linea, err := lector.ReadString('\n')
		texto := strings.ReplaceAll(strings.TrimSpace(linea), ",", ".")
		if n, e := strconv.ParseFloat(texto, 64); e == nil {
			return n
		}
		if err != nil {
			fmt.Println()
			fmt.Println("No hay más datos.")
			os.Exit(1)
		}
		fmt.Println("Eso no es un número. Prueba otra vez.")
	}
}

func main() {
	lector := bufio.NewReader(os.Stdin)
	mayor := leerNumero(lector, "Escribe el número 1: ")
	for i := 2; i <= 3; i++ {
		n := leerNumero(lector, fmt.Sprintf("Escribe el número %d: ", i))
		if n > mayor {
			mayor = n
		}
	}
	fmt.Println("El mayor es", mayor)
}

// plantilla: invertir_texto
// titulo: Programa que escribe un texto al revés
// descripcion: Lee una línea de texto y la escribe al revés, letra a letra (funciona con tildes y eñes).
// palabras: programa, invertir, reves, texto, frase, palabra, leer, vuelta
// caso: {"huecos": {}, "caso": {"entrada": "hola\n", "esperado": "Al revés: aloh", "comparar": "contiene"}}
// caso: {"huecos": {}, "caso": {"entrada": "ñandú\n", "esperado": "Al revés: údnañ", "comparar": "contiene"}}
// caso: {"huecos": {}, "caso": {"entrada": "Go es genial", "esperado": "Al revés: laineg se oG", "comparar": "contiene"}}
// caso: {"huecos": {}, "caso": {"entrada": "", "esperado": "Al revés:", "comparar": "contiene"}}
package main

import (
	"bufio"
	"fmt"
	"os"
	"strings"
)

func main() {
	lector := bufio.NewReader(os.Stdin)
	fmt.Print("Escribe un texto: ")
	linea, _ := lector.ReadString('\n')
	letras := []rune(strings.TrimRight(linea, "\r\n"))
	for i, j := 0, len(letras)-1; i < j; i, j = i+1, j-1 {
		letras[i], letras[j] = letras[j], letras[i]
	}
	fmt.Println()
	fmt.Println("Al revés:", string(letras))
}
