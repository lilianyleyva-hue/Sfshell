// nota: método
// alias: metodo, metodos, métodos, method, receptor
// titulo: Métodos
// resumen: Un método es una función unida a un tipo; se declara con un receptor: func (r Rectangulo) Area() float64. Si el receptor es un puntero (*Rectangulo), el método puede cambiar el valor original.

package main

import "fmt"

type Rectangulo struct {
	Ancho, Alto float64
}

func (r Rectangulo) Area() float64 {
	return r.Ancho * r.Alto
}

func (r *Rectangulo) Escalar(f float64) {
	r.Ancho *= f
	r.Alto *= f
}

func main() {
	r := Rectangulo{3, 4}
	fmt.Println(r.Area())
	r.Escalar(2)
	fmt.Println(r, r.Area())

// nota: interfaz
// alias: interface, interfaces, interfaces de go
// titulo: Interfaces
// resumen: Una interfaz es una lista de métodos. Cualquier tipo que tenga esos métodos la cumple, sin tener que decirlo. Así una función puede trabajar con tipos distintos que se comportan igual.
// relacionados: método, struct, error
// palabras: interfaz, interface, metodos, comportamiento, polimorfismo, tipo
// salida: main.Cuadrado: 4.00
// salida: main.Circulo: 3.14
// salida: Total: 7.14
package main

import (
	"fmt"
	"math"
)

type Figura interface {
	Area() float64
}

type Cuadrado struct{ Lado float64 }

type Circulo struct{ Radio float64 }

func (c Cuadrado) Area() float64 { return c.Lado * c.Lado }

func (c Circulo) Area() float64 { return math.Pi * c.Radio * c.Radio }

func main() {
	figuras := []Figura{Cuadrado{2}, Circulo{1}}
	total := 0.0
	for _, f := range figuras {
		fmt.Printf("%T: %.2f\n", f, f.Area())
		total += f.Area()
	}
	fmt.Printf("Total: %.2f\n", total)
}

// nota: error
// alias: errores, manejo de errores, err, if err != nil
// titulo: Errores
// resumen: En Go los errores son valores: una función que puede fallar devuelve un error como último resultado. Quien la llama lo comprueba con if err != nil y decide qué hacer. nil significa que no hubo error.
// relacionados: errors.New, fmt.Errorf, errors.Is, panic
// palabras: error, fallo, comprobar, devolver, err, nil
// salida: 3.5
// salida: error: división entre cero
// salida: 0 strconv.Atoi: parsing "12a": invalid syntax
package main

import (
	"errors"
	"fmt"
	"strconv"
)

func dividir(a, b float64) (float64, error) {
	if b == 0 {
		return 0, errors.New("división entre cero")
	}
	return a / b, nil
}

func main() {
	if r, err := dividir(7, 2); err == nil {
		fmt.Println(r)
	}
	if _, err := dividir(1, 0); err != nil {
		fmt.Println("error:", err)
	}
	n, err := strconv.Atoi("12a")
	fmt.Println(n, err)
}

// nota: defer
// alias: diferir, defers
// titulo: defer: dejar algo para el final
// resumen: defer apunta una llamada para que se haga cuando termine la función, pase lo que pase (también si hay un pánico). Se usa para cerrar archivos o liberar cerrojos justo después de abrirlos. Si hay varios, se ejecutan al revés: el último apuntado va primero.

// nota: panic/recover
// alias: panico, pánico, excepcion, excepciones
// titulo: Pánicos: panic y recover
// resumen: Un pánico es un error grave que para el programa: un índice fuera de rango, dividir entre cero o una llamada a panic. Dentro de un defer, recover() lo detiene y el programa sigue. Go no tiene excepciones: para errores normales se devuelven valores error.
// relacionados: panic, recover, defer, error
// palabras: panico, recuperar, excepcion, error, grave, parar
// salida: 5 true
// salida: recuperado: runtime error: integer divide by zero
// salida: 0 false
package main

import "fmt"

func dividirSeguro(a, b int) (r int, ok bool) {
	defer func() {
		if e := recover(); e != nil {
			fmt.Println("recuperado:", e)
			ok = false
		}
	}()
	return a / b, true
}

func main() {
	fmt.Println(dividirSeguro(10, 2))
	fmt.Println(dividirSeguro(1, 0))
}

// nota: goroutine
// alias: goroutines, gorutina, gorrutina, go, concurrencia
// titulo: Goroutines
// resumen: Escribir go delante de una llamada la ejecuta a la vez que el resto del programa, en una goroutine. Para recoger sus resultados se usan canales o sync.WaitGroup. El programa termina cuando acaba main, aunque queden goroutines.
// relacionados: canal, sync.WaitGroup, select, sync.Mutex
// palabras: goroutine, concurrencia, paralelo, simultaneo, hilo, go
// salida: 1² = 1, 2² = 4, 3² = 9
package main

import (
	"fmt"
	"sort"
	"strings"
)

func cuadrado(n int, resultados chan<- string) {
	resultados <- fmt.Sprintf("%d² = %d", n, n*n)
}

func main() {
	resultados := make(chan string)
	for i := 1; i <= 3; i++ {
		go cuadrado(i, resultados)
	}
	var todos []string
	for i := 0; i < 3; i++ {
		todos = append(todos, <-resultados)
	}
	sort.Strings(todos) // llegan en cualquier orden
	fmt.Println(strings.Join(todos, ", "))
}

// nota: canal
// alias: canales, chan, channel, channels
// titulo: Canales
// resumen: Un canal es un tubo para pasar valores entre goroutines: c <- v envía y v := <-c recibe. Sin capacidad, enviar espera a que alguien reciba. Quien envía puede cerrarlo con close, y for v := range c lee hasta que se cierra.
// relacionados: goroutine, select, make
// palabras: canal, enviar, recibir, comunicar, goroutine, chan, cerrar
// salida: recibo 10
// salida: recibo 20
// salida: recibo 30
// salida: 2 2
package main

import "fmt"

func main() {
	c := make(chan int)
	go func() {
		for i := 1; i <= 3; i++ {
			c <- i * 10
		}
		close(c)
	}()
	for v := range c {
		fmt.Println("recibo", v)
	}
	b := make(chan string, 2)
	b <- "a"
	b <- "b"
	fmt.Println(len(b), cap(b))
}

// nota: select
// alias: selección de canales
// titulo: select: esperar a varios canales
// resumen: select espera a que uno de varios canales esté listo y ejecuta ese caso. Con default no espera nada si ninguno está listo, y con time.After se pone un tiempo máximo.
// relacionados: canal, goroutine, time.Duration
// palabras: select, esperar, canales, varios, tiempo, maximo, default
// salida: llegó: listo
// salida: no hay nada que leer
// salida: tiempo agotado
package main

import (
	"fmt"
	"time"
)

func main() {
	rapido := make(chan string, 1)
	rapido <- "listo"
	select {
	case m := <-rapido:
		fmt.Println("llegó:", m)
	case <-time.After(time.Second):
		fmt.Println("tiempo agotado")
	}

	vacio := make(chan string)
	select {
	case m := <-vacio:
		fmt.Println(m)
	default:
		fmt.Println("no hay nada que leer")
	}

	lento := make(chan string)
	select {
	case m := <-lento:
		fmt.Println(m)
	case <-time.After(20 * time.Millisecond):
		fmt.Println("tiempo agotado")
	}
}

// nota: genéricos
// alias: genericos, generico, genérico, generics, type parameters, parametros de tipo
// titulo: Genéricos (Go 1.18)
// resumen: Una función genérica tiene parámetros de tipo entre corchetes y sirve para muchos tipos a la vez: func Maximo[T cmp.Ordered](xs []T) T. La restricción (cmp.Ordered, any o una interfaz como ~int | ~float64) dice qué tipos valen.
// relacionados: interfaz, slices.Sort
// palabras: generico, tipo, parametro, cualquier, reutilizar, generics
// salida: 9
// salida: uva
// salida: 4
package main

import (
	"cmp"
	"fmt"
)

func Maximo[T cmp.Ordered](xs []T) T {
	m := xs[0]
	for _, x := range xs[1:] {
		if x > m {
			m = x
		}
	}
	return m
}

type Numero interface {
	~int | ~float64
}

func Sumar[T Numero](xs []T) T {
	var s T
	for _, x := range xs {
		s += x
	}
	return s
}

func main() {
	fmt.Println(Maximo([]int{3, 9, 2}))
	fmt.Println(Maximo([]string{"pera", "uva", "kiwi"}))
	fmt.Println(Sumar([]float64{1.5, 2.5}))
}
