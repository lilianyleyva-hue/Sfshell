// nota: sort.Ints
// titulo: sort.Ints: ordenar números enteros
// resumen: Ordena una lista de int de menor a mayor. Cambia la propia lista (no devuelve otra nueva).
// relacionados: slices.Sort, sort.Strings, sort.Slice
// palabras: ordenar, lista, numero, entero, menor, mayor, sort
// salida: [1 2 3 5 8]
package main

import (
	"fmt"
	"sort"
)

func main() {
	nums := []int{5, 2, 8, 1, 3}
	sort.Ints(nums)
	fmt.Println(nums)
}

// nota: sort.Strings
// titulo: sort.Strings: ordenar textos
// resumen: Ordena una lista de textos alfabéticamente, comparando byte a byte: las mayúsculas van antes que las minúsculas y las letras con tilde van al final.
// relacionados: sort.Ints, slices.Sort
// palabras: ordenar, alfabetico, texto, palabra, lista, sort
// salida: [Zoe ana luis árbol]
package main

import (
	"fmt"
	"sort"
)

func main() {
	nombres := []string{"luis", "ana", "árbol", "Zoe"}
	sort.Strings(nombres)
	fmt.Println(nombres)
}

// nota: sort.Slice
// titulo: sort.Slice: ordenar con tu propio criterio
// resumen: Ordena cualquier lista usando una función que dice si el elemento i va antes que el j. Sirve para ordenar estructuras por un campo o en orden descendente.
// relacionados: sort.Ints, slices.SortFunc
// palabras: ordenar, criterio, comparar, estructura, descendente, campo, funcion
// salida: [{Luis 17} {Ana 30} {Eva 42}]
// salida: [9 4 1]
package main

import (
	"fmt"
	"sort"
)

type Persona struct {
	Nombre string
	Edad   int
}

func main() {
	gente := []Persona{{"Ana", 30}, {"Eva", 42}, {"Luis", 17}}
	sort.Slice(gente, func(i, j int) bool { return gente[i].Edad < gente[j].Edad })
	fmt.Println(gente)
	nums := []int{4, 9, 1}
	sort.Slice(nums, func(i, j int) bool { return nums[i] > nums[j] })
	fmt.Println(nums)
}

// nota: slices.Sort
// titulo: slices.Sort: ordenar cualquier lista (Go 1.21)
// resumen: Ordena de menor a mayor una lista de cualquier tipo que se pueda comparar (números, textos…). Es la forma moderna de sort.Ints y sort.Strings.
// relacionados: sort.Ints, slices.Reverse
// palabras: ordenar, lista, menor, mayor, slices, generico
// salida: [1.5 2 3.25]
// salida: [a b c]
package main

import (
	"fmt"
	"slices"
)

func main() {
	xs := []float64{3.25, 1.5, 2}
	slices.Sort(xs)
	fmt.Println(xs)
	letras := []string{"c", "a", "b"}
	slices.Sort(letras)
	fmt.Println(letras)
}

// nota: slices.Contains
// titulo: slices.Contains: ¿está este valor en la lista?
// resumen: Dice si una lista contiene un valor. Recorre la lista entera, así que dentro de un bucle grande es mejor usar un mapa como conjunto.
// relacionados: slices.Index, map
// palabras: contiene, buscar, lista, estar, valor, pertenece
// salida: true
// salida: false
package main

import (
	"fmt"
	"slices"
)

func main() {
	frutas := []string{"pera", "manzana", "uva"}
	fmt.Println(slices.Contains(frutas, "uva"))
	fmt.Println(slices.Contains(frutas, "kiwi"))
}

// nota: slices.Index
// titulo: slices.Index: la posición de un valor
// resumen: Devuelve la posición de la primera aparición de un valor en una lista, o -1 si no está.
// relacionados: slices.Contains, strings.Index
// palabras: posicion, indice, buscar, lista, donde, valor
// salida: 2
// salida: -1
package main

import (
	"fmt"
	"slices"
)

func main() {
	nums := []int{10, 20, 30, 20}
	fmt.Println(slices.Index(nums, 30))
	fmt.Println(slices.Index(nums, 99))
}

// nota: slices.Max
// titulo: slices.Max: el mayor de una lista
// resumen: Devuelve el elemento más grande de una lista. Si la lista está vacía provoca un pánico, así que compruébalo antes.
// relacionados: slices.Min, max
// palabras: maximo, mayor, lista, grande, elemento
// salida: 9
// salida: pera
package main

import (
	"fmt"
	"slices"
)

func main() {
	fmt.Println(slices.Max([]int{3, 9, 2}))

}

// nota: slices.Min
// titulo: slices.Min: el menor de una lista
// resumen: Devuelve el elemento más pequeño de una lista. Si la lista está vacía provoca un pánico.
// relacionados: slices.Max, min
// palabras: minimo, menor, lista, pequeño, elemento
// salida: -4
package main

import (
	"fmt"
	"slices"
)

func main() {
	temperaturas := []int{3, -4, 12, 0}
	if len(temperaturas) > 0 {
		fmt.Println(slices.Min(temperaturas))
	}
}

// nota: maps.Keys
// titulo: maps.Keys: las claves de un mapa
// resumen: El paquete maps llegó en Go 1.21, pero maps.Keys no se añadió hasta Go 1.23, y devuelve un iterador, no una lista. Para tener las claves ordenadas en una lista usa slices.Sorted(maps.Keys(m)); en versiones antiguas, recórrelas con for range y ordénalas.
// relacionados: map, for/range, slices.Sort
// palabras: claves, mapa, keys, lista, ordenar, recorrer
// salida: [ana eva luis]
// salida: 3
package main

import (
	"fmt"
	"maps"
	"slices"
)

func main() {
	edades := map[string]int{"luis": 17, "ana": 30, "eva": 42}
	claves := slices.Sorted(maps.Keys(edades))
	fmt.Println(claves)
	fmt.Println(len(claves))
}
